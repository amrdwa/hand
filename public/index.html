function renderRow(rowId, cards) { const container = document.getElementById(rowId); container.innerHTML = ""; container.ondragover = (e) => e.preventDefault(); container.ondrop = (e) => { e.preventDefault(); if (!draggedCardId) return; // إذا سحبته على الفراغ في السطر نفسه، بنحطه بالاخر const allCards = [...myHandRow1, ...myHandRow2]; const sourceCard = allCards.find(c => c.id == draggedCardId); if (!sourceCard) return; myHandRow1 = myHandRow1.filter(c => c.id != draggedCardId); myHandRow2 = myHandRow2.filter(c => c.id != draggedCardId); if (rowId === "row1") myHandRow1.push(sourceCard); else myHandRow2.push(sourceCard); draggedCardId = null; renderHand(); }; cards.forEach((card) => { const div = document.createElement("div"); const isSelected = selectedCardId === card.id; div.className = `card ${card.color} ${isSelected ? 'selected' : ''}`; div.dataset.id = card.id; div.draggable = true; div.innerHTML = `
${card.value}
${card.suit}
`; div.onclick = (e) => { e.stopPropagation(); selectedCardId = isSelected ? null : card.id; renderHand(); }; div.ondragstart = (e) => { draggedCardId = card.id; div.classList.add("dragging"); e.dataTransfer.setData("text/plain", card.id); }; div.ondragend = () => { div.classList.remove("dragging"); clearDropTargetStyles(); }; div.ondragover = (e) => { e.preventDefault(); e.stopPropagation(); if (draggedCardId !== card.id) { div.classList.add("drop-target-left"); } }; div.ondragleave = () => { div.classList.remove("drop-target-left"); }; div.ondrop = (e) => { e.preventDefault(); e.stopPropagation(); clearDropTargetStyles(); if (draggedCardId && draggedCardId !== card.id) { const rect = div.getBoundingClientRect(); const offsetX = e.clientX - rect.left; // موقع الماوس بالنسبة لعرض الكرت // إذا أفلتته على النصف الأيمن من الكرت (حسب الـ RTL يعني جهة السبعة)، بنحطه قبله تماماً // وإذا على النصف الأيسر بنحطه بعده const insertBefore = offsetX > (rect.width / 2); moveCardToTargetWithPrecision(draggedCardId, card.id, rowId, insertBefore); } }; container.appendChild(div); }); } function moveCardToTargetWithPrecision(sourceId, targetId, targetRowId, insertBefore) { const allCards = [...myHandRow1, ...myHandRow2]; const sourceCard = allCards.find(c => c.id == sourceId); if (!sourceCard) return; myHandRow1 = myHandRow1.filter(c => c.id != sourceId); myHandRow2 = myHandRow2.filter(c => c.id != sourceId); const targetRow = targetRowId === "row1" ? myHandRow1 : myHandRow2; const targetIdx = targetRow.findIndex(c => c.id == targetId); if (targetIdx !== -1) { const finalIdx = insertBefore ? targetIdx : targetIdx + 1; targetRow.splice(finalIdx, 0, sourceCard); } else { targetRow.push(sourceCard); } draggedCardId = null; renderHand(); }
